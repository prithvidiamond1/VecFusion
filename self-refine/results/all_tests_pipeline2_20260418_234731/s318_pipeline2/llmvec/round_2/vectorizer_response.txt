#include <stdint.h>
#include <string.h>
#include <math.h>

float vectorized_s318(int iterations, int LEN_1D, float* a, int inc) {
    float max = 0.0f;
    int index = 0;
    
    for (int nl = 0; nl < iterations/2; nl++) {
        int k = 0;
        index = 0;
        max = fabsf(a[0]);
        k += inc;
        
        // Vectorized maximum reduction with index tracking
        float max_val = max;
        int max_idx = 0;
        int current_k = k;
        
        // Process in chunks of 4
        int i;
        for (i = 1; i + 3 < LEN_1D; i += 4) {
            // Load 4 values with stride
            float v0 = fabsf(a[current_k]);
            current_k += inc;
            float v1 = fabsf(a[current_k]);
            current_k += inc;
            float v2 = fabsf(a[current_k]);
            current_k += inc;
            float v3 = fabsf(a[current_k]);
            current_k += inc;
            
            // Track maximum and index
            if (v0 > max_val) { max_val = v0; max_idx = i; }
            if (v1 > max_val) { max_val = v1; max_idx = i + 1; }
            if (v2 > max_val) { max_val = v2; max_idx = i + 2; }
            if (v3 > max_val) { max_val = v3; max_idx = i + 3; }
        }
        
        // Scalar tail
        for (; i < LEN_1D; i++) {
            float abs_val = fabsf(a[current_k]);
            current_k += inc;
            if (abs_val > max_val) {
                max_val = abs_val;
                max_idx = i;
            }
        }
        
        // Update k to match original scalar behavior
        k = current_k;
        
        // Update original variables
        index = max_idx;
        max = max_val;
        float chksum = max + (float)index;
    }
    
    return max + index + 1;
}
