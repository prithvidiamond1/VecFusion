void vectorized_s241(int iterations, int LEN_1D, float* a, float* b, float* c, float* d)
{
    for (int nl = 0; nl < 2 * iterations; nl++) {
        int i;
        int limit = LEN_1D - 1;

        // Single fused loop matching original semantics:
        // a[i] = b[i]*c[i]*d[i], then b[i] = a[i]*a[i+1]*d[i]
        // where a[i+1] is the OLD value (not yet updated at iteration i)
        // Unroll by 4 for vectorization hints
        int vec_limit = limit - (limit % 4);
        for (i = 0; i < vec_limit; i += 4) {
            float a_old_i1 = a[i+1];
            float a_old_i2 = a[i+2];
            float a_old_i3 = a[i+3];
            float a_old_i4 = a[i+4];

            float new_a0 = b[i+0] * c[i+0] * d[i+0];
            float new_a1 = b[i+1] * c[i+1] * d[i+1];
            float new_a2 = b[i+2] * c[i+2] * d[i+2];
            float new_a3 = b[i+3] * c[i+3] * d[i+3];

            a[i+0] = new_a0;
            a[i+1] = new_a1;
            a[i+2] = new_a2;
            a[i+3] = new_a3;

            b[i+0] = new_a0 * a_old_i1 * d[i+0];
            b[i+1] = new_a1 * a_old_i2 * d[i+1];
            b[i+2] = new_a2 * a_old_i3 * d[i+2];
            b[i+3] = new_a3 * a_old_i4 * d[i+3];
        }
        for (; i < limit; i++) {
            float old_next = a[i+1];
            a[i] = b[i] * c[i] * d[i];
            b[i] = a[i] * old_next * d[i];
        }
    }
}
