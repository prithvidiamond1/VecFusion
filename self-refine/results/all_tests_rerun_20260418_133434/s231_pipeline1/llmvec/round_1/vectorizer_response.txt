void vectorized_s231(int iterations, float aa[256][256], float bb[256][256])
{
    for (int nl = 0; nl < 100*(iterations/256); nl++) {
        for (int i = 0; i < 256; i += 8) {
            int iend = i + 8 < 256 ? i + 8 : 256;
            for (int j = 1; j < 256; j++) {
                for (int ii = i; ii < iend; ii++) {
                    aa[j][ii] = aa[j - 1][ii] + bb[j][ii];
                }
            }
        }
    }
}
