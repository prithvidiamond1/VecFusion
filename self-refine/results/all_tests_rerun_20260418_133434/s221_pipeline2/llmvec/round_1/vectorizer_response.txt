#include <stdint.h>

void vectorized_s221(int iterations, int LEN_1D, float*a, float*b, float*c, float*d)
{
    for (int nl = 0; nl < iterations/2; nl++) {
        // First loop: independent parallel computation - vectorizable
        int i = 1;
        int len = LEN_1D - 1; // number of elements from index 1

        // Vector width of 8 floats
        #define VW 8
        int vec_end = 1 + (len / VW) * VW;

        for (; i < vec_end; i += VW) {
            a[i+0] += c[i+0] * d[i+0];
            a[i+1] += c[i+1] * d[i+1];
            a[i+2] += c[i+2] * d[i+2];
            a[i+3] += c[i+3] * d[i+3];
            a[i+4] += c[i+4] * d[i+4];
            a[i+5] += c[i+5] * d[i+5];
            a[i+6] += c[i+6] * d[i+6];
            a[i+7] += c[i+7] * d[i+7];
        }
        // Scalar tail
        for (; i < LEN_1D; i++) {
            a[i] += c[i] * d[i];
        }

        // Second loop: sequential recurrence - cannot be vectorized
        float b_prev = b[0];
        for (i = 1; i < LEN_1D; i++) {
            float temp = b_prev + a[i] + d[i];
            b_prev = temp;
            b[i] = temp;
        }
        #undef VW
    }
}
