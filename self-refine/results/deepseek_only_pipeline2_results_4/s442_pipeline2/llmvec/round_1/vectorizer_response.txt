#include <stdint.h>
#include <string.h>

void vectorized_s442(int iterations, int LEN_1D, float* a, float* b, float* c, float* d, float* e, int* indx) {
    int outer_iters = iterations / 2;
    
    for (int nl = 0; nl < outer_iters; nl++) {
        // Process all cases in a single vectorized loop
        int i = 0;
        
        // Vectorized main loop
        for (; i + 3 < LEN_1D; i += 4) {
            // Load indices
            int idx0 = indx[i];
            int idx1 = indx[i + 1];
            int idx2 = indx[i + 2];
            int idx3 = indx[i + 3];
            
            // Load a values
            float a0 = a[i];
            float a1 = a[i + 1];
            float a2 = a[i + 2];
            float a3 = a[i + 3];
            
            // Process case 1
            if (idx0 == 1) a0 += b[i] * b[i];
            if (idx1 == 1) a1 += b[i + 1] * b[i + 1];
            if (idx2 == 1) a2 += b[i + 2] * b[i + 2];
            if (idx3 == 1) a3 += b[i + 3] * b[i + 3];
            
            // Process case 2
            if (idx0 == 2) a0 += c[i] * c[i];
            if (idx1 == 2) a1 += c[i + 1] * c[i + 1];
            if (idx2 == 2) a2 += c[i + 2] * c[i + 2];
            if (idx3 == 2) a3 += c[i + 3] * c[i + 3];
            
            // Process case 3
            if (idx0 == 3) a0 += d[i] * d[i];
            if (idx1 == 3) a1 += d[i + 1] * d[i + 1];
            if (idx2 == 3) a2 += d[i + 2] * d[i + 2];
            if (idx3 == 3) a3 += d[i + 3] * d[i + 3];
            
            // Process case 4
            if (idx0 == 4) a0 += e[i] * e[i];
            if (idx1 == 4) a1 += e[i + 1] * e[i + 1];
            if (idx2 == 4) a2 += e[i + 2] * e[i + 2];
            if (idx3 == 4) a3 += e[i + 3] * e[i + 3];
            
            // Store results
            a[i] = a0;
            a[i + 1] = a1;
            a[i + 2] = a2;
            a[i + 3] = a3;
        }
        
        // Scalar cleanup for remaining elements
        for (; i < LEN_1D; i++) {
            int idx = indx[i];
            if (idx == 1) {
                a[i] += b[i] * b[i];
            } else if (idx == 2) {
                a[i] += c[i] * c[i];
            } else if (idx == 3) {
                a[i] += d[i] * d[i];
            } else if (idx == 4) {
                a[i] += e[i] * e[i];
            }
        }
    }
}
