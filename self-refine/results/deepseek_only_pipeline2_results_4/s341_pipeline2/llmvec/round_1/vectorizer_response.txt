#include <stddef.h>

void vectorized_s341(int iterations, int LEN_1D, float* a, float* b) {
    for (int nl = 0; nl < iterations; nl++) {
        int j = 0;
        int i = 0;
        
        // Process in chunks of 4 for simple vectorization
        for (; i + 3 < LEN_1D; i += 4) {
            float b0 = b[i];
            float b1 = b[i + 1];
            float b2 = b[i + 2];
            float b3 = b[i + 3];
            
            // Check each element and store if positive
            if (b0 > 0.0f) {
                a[j] = b0;
                j++;
            }
            if (b1 > 0.0f) {
                a[j] = b1;
                j++;
            }
            if (b2 > 0.0f) {
                a[j] = b2;
                j++;
            }
            if (b3 > 0.0f) {
                a[j] = b3;
                j++;
            }
        }
        
        // Scalar tail
        for (; i < LEN_1D; i++) {
            float temp = b[i];
            if (temp > 0.0f) {
                a[j] = temp;
                j++;
            }
        }
    }
}
