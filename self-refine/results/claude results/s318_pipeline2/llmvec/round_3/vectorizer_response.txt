#include <math.h>
#include <stdint.h>

float vectorized_s318(int iterations, int LEN_1D, float* a, int inc)
{
    int index;
    float max, chksum;
    chksum = 0.0f;
    max = 0.0f;
    index = 0;

    for (int nl = 0; nl < iterations; nl++) {
        index = 0;
        max = fabsf(a[0]);

        int i = 1;
        for (; i <= LEN_1D - 4; i += 4) {
            float v0 = fabsf(a[i * inc]);
            float v1 = fabsf(a[(i+1) * inc]);
            float v2 = fabsf(a[(i+2) * inc]);
            float v3 = fabsf(a[(i+3) * inc]);
            if (v0 > max) { max = v0; index = i; }
            if (v1 > max) { max = v1; index = i+1; }
            if (v2 > max) { max = v2; index = i+2; }
            if (v3 > max) { max = v3; index = i+3; }
        }
        for (; i < LEN_1D; i++) {
            float val = fabsf(a[i * inc]);
            if (val > max) { max = val; index = i; }
        }

        chksum = max + (float)index;
    }
    return chksum;
}
