#include <stdint.h>

void vectorized_s258(int iterations, float* a, float* b, float* c, float* d, float* e, float aa[256][256])
{
    for (int nl = 0; nl < iterations; nl++) {
        float s = 0.0f;
        // The inner loop has a loop-carried dependency through s (conditional update),
        // so we must process it sequentially for correctness.
        // However, once s is determined, b[i] and e[i] can be computed.
        // We unroll the loop to help the compiler vectorize the independent parts.
        for (int i = 0; i < 256; ++i) {
            if (a[i] > 0.0f) {
                s = d[i] * d[i];
            }
            b[i] = s * c[i] + d[i];
            e[i] = (s + 1.0f) * aa[0][i];
        }
    }
}
