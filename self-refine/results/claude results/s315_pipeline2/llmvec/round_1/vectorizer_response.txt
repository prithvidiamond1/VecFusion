#include <stdint.h>
#include <float.h>

float vectorized_s315(int iterations, int LEN_1D, float* a)
{
    for (int i = 0; i < LEN_1D; i++)
        a[i] = (i * 7) % LEN_1D;

    float x, chksum;
    int index;

    for (int nl = 0; nl < iterations; nl++) {
        /* First pass: find the maximum value using unrolled reduction */
        float max0 = -FLT_MAX, max1 = -FLT_MAX, max2 = -FLT_MAX, max3 = -FLT_MAX;

        int i = 0;
        int limit = LEN_1D - (LEN_1D % 4);
        for (; i < limit; i += 4) {
            float v0 = a[i+0];
            float v1 = a[i+1];
            float v2 = a[i+2];
            float v3 = a[i+3];
            max0 = v0 > max0 ? v0 : max0;
            max1 = v1 > max1 ? v1 : max1;
            max2 = v2 > max2 ? v2 : max2;
            max3 = v3 > max3 ? v3 : max3;
        }
        for (; i < LEN_1D; i++) {
            float v = a[i];
            max0 = v > max0 ? v : max0;
        }
        float tmp0 = max0 > max1 ? max0 : max1;
        float tmp1 = max2 > max3 ? max2 : max3;
        float max_val = tmp0 > tmp1 ? tmp0 : tmp1;

        /* Second pass: find minimum index where a[i] == max_val */
        /* Use unrolled lanes tracking neg_idx (max of -i where a[i]==max_val) */
        int neg0 = -LEN_1D, neg1 = -LEN_1D, neg2 = -LEN_1D, neg3 = -LEN_1D;

        i = 0;
        for (; i < limit; i += 4) {
            int c0 = (a[i+0] == max_val);
            int c1 = (a[i+1] == max_val);
            int c2 = (a[i+2] == max_val);
            int c3 = (a[i+3] == max_val);
            int cand0 = -(i+0);
            int cand1 = -(i+1);
            int cand2 = -(i+2);
            int cand3 = -(i+3);
            if (c0 & (cand0 > neg0)) neg0 = cand0;
            if (c1 & (cand1 > neg1)) neg1 = cand1;
            if (c2 & (cand2 > neg2)) neg2 = cand2;
            if (c3 & (cand3 > neg3)) neg3 = cand3;
        }
        for (; i < LEN_1D; i++) {
            int c = (a[i] == max_val);
            int cand = -i;
            if (c & (cand > neg0)) neg0 = cand;
        }
        /* Merge lanes: pick the maximum neg_idx (= minimum index) */
        int n01 = neg0 > neg1 ? neg0 : neg1;
        int n23 = neg2 > neg3 ? neg2 : neg3;
        int neg_idx = n01 > n23 ? n01 : n23;

        x = max_val;
        index = -neg_idx;
        chksum = x + (float)index;
        (void)chksum;
    }
    return index + x + 1;
}
