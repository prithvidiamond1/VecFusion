#include <stdint.h>

void vectorized_s258(int iterations, float* a, float* b, float* c, float* d, float* e, float* aa)
{
    float s = 0.0f;
    for (int nl = 0; nl < iterations; nl++) {
        s = 0.0f;
        for (int i = 0; i < 256; ++i) {
            if (a[i] > 0.0f) {
                s = d[i] * d[i];
            }
            b[i] = s * c[i] + d[i];
            e[i] = (s + 1.0f) * aa[i];
        }
    }
}
