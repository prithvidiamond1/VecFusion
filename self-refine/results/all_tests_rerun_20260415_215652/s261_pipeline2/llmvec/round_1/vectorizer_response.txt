#include <stdint.h>

void vectorized_s261(int iterations, int LEN_1D, float* a, float* b, float* c, float* d) {
    for (int nl = 0; nl < iterations; nl++) {
        // Vectorize: c[i] = c[i] * d[i] for i in [1, LEN_1D)
        int i = 1;
        int limit = LEN_1D;
        int vec_end = 1 + ((limit - 1) / 8) * 8;

        for (; i + 7 < limit; i += 8) {
            c[i+0] = c[i+0] * d[i+0];
            c[i+1] = c[i+1] * d[i+1];
            c[i+2] = c[i+2] * d[i+2];
            c[i+3] = c[i+3] * d[i+3];
            c[i+4] = c[i+4] * d[i+4];
            c[i+5] = c[i+5] * d[i+5];
            c[i+6] = c[i+6] * d[i+6];
            c[i+7] = c[i+7] * d[i+7];
        }
        for (; i < limit; i++) {
            c[i] = c[i] * d[i];
        }

        // Vectorize: a[i] = a[i] + b[i] + c[i-1] for i in [1, LEN_1D)
        i = 1;
        for (; i + 7 < limit; i += 8) {
            a[i+0] = a[i+0] + b[i+0] + c[i+0-1];
            a[i+1] = a[i+1] + b[i+1] + c[i+1-1];
            a[i+2] = a[i+2] + b[i+2] + c[i+2-1];
            a[i+3] = a[i+3] + b[i+3] + c[i+3-1];
            a[i+4] = a[i+4] + b[i+4] + c[i+4-1];
            a[i+5] = a[i+5] + b[i+5] + c[i+5-1];
            a[i+6] = a[i+6] + b[i+6] + c[i+6-1];
            a[i+7] = a[i+7] + b[i+7] + c[i+7-1];
        }
        for (; i < limit; i++) {
            a[i] = a[i] + b[i] + c[i-1];
        }
    }
}
