#include <stdint.h>

void vectorized_s2233(int iterations, float aa[256][256], float bb[256][256], float cc[256][256])
{
    for (int nl = 0; nl < 100*(iterations/256); nl++) {
        for (int i = 1; i < 256; i++) {
            // Loop 1: aa[j][i] = aa[j-1][i] + cc[j][i]
            // Each iteration depends on previous j, so this is a serial recurrence
            // Cannot vectorize across j due to dependency, but we can unroll slightly
            for (int j = 1; j < 256; j++) {
                aa[j][i] = aa[j-1][i] + cc[j][i];
            }

            // Loop 2: bb[i][j] = bb[i-1][j] + cc[i][j]
            // No dependency across j, can vectorize
            {
                int j = 1;
                // Process 8 elements at a time
                for (; j <= 255 - 7; j += 8) {
                    bb[i][j+0] = bb[i-1][j+0] + cc[i][j+0];
                    bb[i][j+1] = bb[i-1][j+1] + cc[i][j+1];
                    bb[i][j+2] = bb[i-1][j+2] + cc[i][j+2];
                    bb[i][j+3] = bb[i-1][j+3] + cc[i][j+3];
                    bb[i][j+4] = bb[i-1][j+4] + cc[i][j+4];
                    bb[i][j+5] = bb[i-1][j+5] + cc[i][j+5];
                    bb[i][j+6] = bb[i-1][j+6] + cc[i][j+6];
                    bb[i][j+7] = bb[i-1][j+7] + cc[i][j+7];
                }
                // Scalar cleanup
                for (; j < 256; j++) {
                    bb[i][j] = bb[i-1][j] + cc[i][j];
                }
            }
        }
    }
}
