#include <stdbool.h>
#include <stdint.h>
#include <string.h>

void vectorized_AllPointsEq(const int pts[], int count, bool *out_vector, int n) {
    (void)n;

    if (count <= 1) {
        out_vector[0] = true;
        return;
    }

    int val = pts[0];

    typedef int v8si __attribute__((vector_size(32)));

    v8si vval = {val, val, val, val, val, val, val, val};

    int i = 1;
    int simd_count = (count - 1) / 8;
    int simd_end = 1 + simd_count * 8;

    for (; i < simd_end; i += 8) {
        v8si chunk;
        memcpy(&chunk, &pts[i], sizeof(v8si));
        v8si cmp = (chunk != vval);
        int r = cmp[0] | cmp[1] | cmp[2] | cmp[3] |
                cmp[4] | cmp[5] | cmp[6] | cmp[7];
        if (r) {
            out_vector[0] = false;
            return;
        }
    }

    /* Scalar tail */
    for (; i < count; ++i) {
        if (pts[i] != val) {
            out_vector[0] = false;
            return;
        }
    }

    out_vector[0] = true;
}
