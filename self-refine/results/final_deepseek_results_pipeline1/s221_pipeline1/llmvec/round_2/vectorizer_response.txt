#include <stddef.h>

typedef float float_vec __attribute__((__vector_size__(16)));

void vectorized_s221(int iterations, int LEN_1D, float *a, float *b, float *c, float *d) {
    int total_iters = iterations;
    for (int nl = 0; nl < total_iters; nl++) {
        int i = 1;
        a[1] += c[1] * d[1];
        b[1] = b[0] + a[1] + d[1];
        i = 2;

        for (; i + 3 < LEN_1D; i += 4) {
            float_vec va = __builtin_assume_aligned(&a[i], 16) ? *(float_vec *)&a[i] : *(float_vec *)&a[i];
            float_vec vc = __builtin_assume_aligned(&c[i], 16) ? *(float_vec *)&c[i] : *(float_vec *)&c[i];
            float_vec vd = __builtin_assume_aligned(&d[i], 16) ? *(float_vec *)&d[i] : *(float_vec *)&d[i];
            float_vec vb_prev = __builtin_assume_aligned(&b[i - 1], 16) ? *(float_vec *)&b[i - 1] : *(float_vec *)&b[i - 1];

            va = va + vc * vd;
            if (__builtin_assume_aligned(&a[i], 16)) *(float_vec *)&a[i] = va;

            float temp_b = vb_prev[3] + va[0] + vd[0];
            b[i] = temp_b;
            temp_b = temp_b + va[1] + vd[1];
            b[i + 1] = temp_b;
            temp_b = temp_b + va[2] + vd[2];
            b[i + 2] = temp_b;
            temp_b = temp_b + va[3] + vd[3];
            b[i + 3] = temp_b;
        }

        for (; i < LEN_1D; i++) {
            a[i] += c[i] * d[i];
            b[i] = b[i - 1] + a[i] + d[i];
        }
    }
}
