#include <stddef.h>

void vectorized_s222(int iterations, int LEN_1D, float *a, float *b, float *c, float *e) {
    if (iterations <= 0 || LEN_1D <= 1) return;

    for (int nl = 0; nl < iterations; nl++) {
        // Process index 0 separately to initialize recurrence
        float e_prev = e[0];
        float bc0 = b[0] * c[0];
        a[0] += bc0;
        float e_sq0 = e_prev * e_prev;
        e[0] = e_sq0;
        a[0] -= bc0;
        e_prev = e_sq0;

        // Process remaining indices with scalar recurrence
        for (int i = 1; i < LEN_1D; i++) {
            float bc = b[i] * c[i];
            a[i] += bc;
            float e_sq = e_prev * e_prev;
            e[i] = e_sq;
            a[i] -= bc;
            e_prev = e_sq;
        }
    }
}
