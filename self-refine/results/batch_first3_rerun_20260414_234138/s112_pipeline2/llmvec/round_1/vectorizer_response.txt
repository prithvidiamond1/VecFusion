#include <stdlib.h>
#include <string.h>

void vectorized_s112(float *a, float *b, int iterations, int LEN_1D) {
    float *temp = (float *)malloc(LEN_1D * sizeof(float));
    for (int nl = 0; nl < 3 * iterations; nl++) {
        memcpy(temp, a, LEN_1D * sizeof(float));
        int i = 0;
        int limit = LEN_1D - 1;
        /* Unroll by 8 for vector-friendly code */
        int limit8 = limit - (limit % 8);
        for (; i < limit8; i += 8) {
            a[i + 1] = temp[i]     + b[i];
            a[i + 2] = temp[i + 1] + b[i + 1];
            a[i + 3] = temp[i + 2] + b[i + 2];
            a[i + 4] = temp[i + 3] + b[i + 3];
            a[i + 5] = temp[i + 4] + b[i + 4];
            a[i + 6] = temp[i + 5] + b[i + 5];
            a[i + 7] = temp[i + 6] + b[i + 6];
            a[i + 8] = temp[i + 7] + b[i + 7];
        }
        /* Scalar cleanup */
        for (; i < limit; i++) {
            a[i + 1] = temp[i] + b[i];
        }
    }
    free(temp);
}
