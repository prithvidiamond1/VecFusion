    for (int nl = 0; nl < iterations/2; nl++) {
        // Split: compute a[i] update first (independent, vectorizable)
        for (int i = 1; i < LEN_1D; i++) {
            a[i] += c[i] * d[i];
        }
        // Then compute b[i] sequentially (carried dependency preserved)
        for (int i = 1; i < LEN_1D; i++) {
            b[i] = b[i - 1] + a[i] + d[i];
        }
    }
}
