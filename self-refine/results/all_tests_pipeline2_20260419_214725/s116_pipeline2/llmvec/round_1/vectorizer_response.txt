#include <stddef.h>

void vectorized_s116(int iterations, int LEN_1D, float* a) {
    if (LEN_1D < 6) return;
    
    int outer_iterations = iterations * 10;
    int limit = LEN_1D - 5;
    
    for (int nl = 0; nl < outer_iterations; nl++) {
        int i = 0;
        
        // Vectorized main loop: process 5 elements at a time
        for (; i + 5 <= limit; i += 5) {
            // Load original values
            float orig_i = a[i];
            float orig_i1 = a[i + 1];
            float orig_i2 = a[i + 2];
            float orig_i3 = a[i + 3];
            float orig_i4 = a[i + 4];
            float orig_i5 = a[i + 5];
            
            // Compute results using original values
            a[i] = orig_i1 * orig_i;
            a[i + 1] = orig_i2 * orig_i1;
            a[i + 2] = orig_i3 * orig_i2;
            a[i + 3] = orig_i4 * orig_i3;
            a[i + 4] = orig_i5 * orig_i4;
        }
        
        // Scalar cleanup for remaining elements
        for (; i < limit; i++) {
            float orig_i = a[i];
            float orig_i1 = a[i + 1];
            a[i] = orig_i1 * orig_i;
        }
    }
}
