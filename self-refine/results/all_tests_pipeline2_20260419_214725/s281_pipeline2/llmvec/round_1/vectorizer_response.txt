#include <stddef.h>

void vectorized_s281(int iterations, int LEN_1D, float* a, float* b, float* c) {
    if (LEN_1D <= 0 || iterations <= 0) return;
    
    int half = LEN_1D / 2;
    int second_half_start = half;
    int second_half_end = LEN_1D;
    
    for (int nl = 0; nl < iterations; nl++) {
        // Process second half first (i = half to LEN_1D-1)
        int i = second_half_start;
        for (; i + 3 < second_half_end; i += 4) {
            // Vectorized computation for 4 elements
            float b0 = b[i];
            float b1 = b[i+1];
            float b2 = b[i+2];
            float b3 = b[i+3];
            
            float c0 = c[i];
            float c1 = c[i+1];
            float c2 = c[i+2];
            float c3 = c[i+3];
            
            int idx0 = LEN_1D - i - 1;
            int idx1 = LEN_1D - (i+1) - 1;
            int idx2 = LEN_1D - (i+2) - 1;
            int idx3 = LEN_1D - (i+3) - 1;
            
            float a_src0 = a[idx0];
            float a_src1 = a[idx1];
            float a_src2 = a[idx2];
            float a_src3 = a[idx3];
            
            float x0 = a_src0 + b0 * c0;
            float x1 = a_src1 + b1 * c1;
            float x2 = a_src2 + b2 * c2;
            float x3 = a_src3 + b3 * c3;
            
            a[i] = x0 - 1.0f;
            a[i+1] = x1 - 1.0f;
            a[i+2] = x2 - 1.0f;
            a[i+3] = x3 - 1.0f;
            
            b[i] = x0;
            b[i+1] = x1;
            b[i+2] = x2;
            b[i+3] = x3;
        }
        // Scalar tail for second half
        for (; i < second_half_end; i++) {
            float x = a[LEN_1D - i - 1] + b[i] * c[i];
            a[i] = x - 1.0f;
            b[i] = x;
        }
        
        // Process first half (i = 0 to half-1)
        i = 0;
        for (; i + 3 < half; i += 4) {
            // Vectorized computation for 4 elements
            float b0 = b[i];
            float b1 = b[i+1];
            float b2 = b[i+2];
            float b3 = b[i+3];
            
            float c0 = c[i];
            float c1 = c[i+1];
            float c2 = c[i+2];
            float c3 = c[i+3];
            
            int idx0 = LEN_1D - i - 1;
            int idx1 = LEN_1D - (i+1) - 1;
            int idx2 = LEN_1D - (i+2) - 1;
            int idx3 = LEN_1D - (i+3) - 1;
            
            float a_src0 = a[idx0];
            float a_src1 = a[idx1];
            float a_src2 = a[idx2];
            float a_src3 = a[idx3];
            
            float x0 = a_src0 + b0 * c0;
            float x1 = a_src1 + b1 * c1;
            float x2 = a_src2 + b2 * c2;
            float x3 = a_src3 + b3 * c3;
            
            a[i] = x0 - 1.0f;
            a[i+1] = x1 - 1.0f;
            a[i+2] = x2 - 1.0f;
            a[i+3] = x3 - 1.0f;
            
            b[i] = x0;
            b[i+1] = x1;
            b[i+2] = x2;
            b[i+3] = x3;
        }
        // Scalar tail for first half
        for (; i < half; i++) {
            float x = a[LEN_1D - i - 1] + b[i] * c[i];
            a[i] = x - 1.0f;
            b[i] = x;
        }
    }
}
