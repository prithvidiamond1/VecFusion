#include <stdint.h>

void vectorized_s323(int iterations, int LEN_1D, float* a, float* b, float* c, float* d, float* e) {
    int outer_iters = iterations / 2;
    
    for (int nl = 0; nl < outer_iters; nl++) {
        float b_prev = b[0];
        
        // Process all elements with scalar loop to maintain correct recurrence
        for (int i = 1; i < LEN_1D; i++) {
            float temp_a = b_prev + c[i] * d[i];
            float temp_b = temp_a + c[i] * e[i];
            a[i] = temp_a;
            b[i] = temp_b;
            b_prev = temp_b;
        }
    }
}
