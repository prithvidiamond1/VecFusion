#include <stdint.h>

void vectorized_s211(int iterations, int LEN_1D, float* a, float* b, float* c, float* d, float* e) {
    // Early exit for small arrays
    if (LEN_1D <= 2) return;
    
    for (int nl = 0; nl < iterations; nl++) {
        int loop_bound = LEN_1D - 1;
        
        // Vectorized store of b[i+1] values
        int i = 1;
        for (; i + 3 < loop_bound; i += 4) {
            b[i] = b[i+1];
            b[i+1] = b[i+2];
            b[i+2] = b[i+3];
            b[i+3] = b[i+4];
        }
        // Scalar tail for b[i+1] store
        for (; i < loop_bound; i++) {
            b[i] = b[i+1];
        }
        
        // Vectorized computation: b[i] = b[i] - e[i] * d[i]
        i = 1;
        for (; i + 3 < loop_bound; i += 4) {
            b[i] = b[i] - e[i] * d[i];
            b[i+1] = b[i+1] - e[i+1] * d[i+1];
            b[i+2] = b[i+2] - e[i+2] * d[i+2];
            b[i+3] = b[i+3] - e[i+3] * d[i+3];
        }
        // Scalar tail for b[i] update
        for (; i < loop_bound; i++) {
            b[i] = b[i] - e[i] * d[i];
        }
        
        // Vectorized computation: a[i] = b[i-1] + c[i] * d[i]
        i = 1;
        for (; i + 3 < loop_bound; i += 4) {
            a[i] = b[i-1] + c[i] * d[i];
            a[i+1] = b[i] + c[i+1] * d[i+1];
            a[i+2] = b[i+1] + c[i+2] * d[i+2];
            a[i+3] = b[i+2] + c[i+3] * d[i+3];
        }
        // Scalar tail for a[i] computation
        for (; i < loop_bound; i++) {
            a[i] = b[i-1] + c[i] * d[i];
        }
    }
}
