#include <stddef.h>

void vectorized_s323(int iterations, int LEN_1D, float* a, float* b, float* c, float* d, float* e) {
    int loop_count = iterations / 2;
    
    for (int nl = 0; nl < loop_count; nl++) {
        // Prefix sum for b - vectorizable with careful handling
        float b_prev = b[0];
        int i = 1;
        
        // Vectorized prefix sum computation
        for (; i + 3 < LEN_1D; i += 4) {
            // Compute d[i] + e[i] for 4 elements
            float d_plus_e_0 = d[i] + e[i];
            float d_plus_e_1 = d[i+1] + e[i+1];
            float d_plus_e_2 = d[i+2] + e[i+2];
            float d_plus_e_3 = d[i+3] + e[i+3];
            
            // Multiply by c[i]
            float prod_0 = c[i] * d_plus_e_0;
            float prod_1 = c[i+1] * d_plus_e_1;
            float prod_2 = c[i+2] * d_plus_e_2;
            float prod_3 = c[i+3] * d_plus_e_3;
            
            // Add to previous b value (prefix sum)
            b[i] = b_prev + prod_0;
            b_prev = b[i];
            
            b[i+1] = b_prev + prod_1;
            b_prev = b[i+1];
            
            b[i+2] = b_prev + prod_2;
            b_prev = b[i+2];
            
            b[i+3] = b_prev + prod_3;
            b_prev = b[i+3];
        }
        
        // Scalar tail for prefix sum
        for (; i < LEN_1D; i++) {
            b[i] = b_prev + c[i] * (d[i] + e[i]);
            b_prev = b[i];
        }
        
        // Compute a[1] separately
        a[1] = b[0] + c[1] * d[1];
        
        // Vectorized computation for a[i] = b[i-1] + c[i] * d[i]
        i = 2;
        for (; i + 3 < LEN_1D; i += 4) {
            // Load b[i-1] values
            float b_prev_0 = b[i-1];
            float b_prev_1 = b[i];
            float b_prev_2 = b[i+1];
            float b_prev_3 = b[i+2];
            
            // Compute c[i] * d[i]
            float prod_0 = c[i] * d[i];
            float prod_1 = c[i+1] * d[i+1];
            float prod_2 = c[i+2] * d[i+2];
            float prod_3 = c[i+3] * d[i+3];
            
            // Store results
            a[i] = b_prev_0 + prod_0;
            a[i+1] = b_prev_1 + prod_1;
            a[i+2] = b_prev_2 + prod_2;
            a[i+3] = b_prev_3 + prod_3;
        }
        
        // Scalar tail for a computation
        for (; i < LEN_1D; i++) {
            a[i] = b[i-1] + c[i] * d[i];
        }
    }
}
