typedef float float4 __attribute__((ext_vector_type(4)));

float vectorized_s316(int iterations, int LEN_1D, float* a)
{
    float x = 0.0f;
    
    for (int nl = 0; nl < iterations * 5; nl++) {
        float min_val = a[0];
        
        // Vectorized reduction
        int i = 1;
        int vec_limit = ((LEN_1D - 1) / 4) * 4 + 1;
        
        if (vec_limit > 1) {
            float4 min_vec = (float4){a[0], a[0], a[0], a[0]};
            
            for (; i < vec_limit; i += 4) {
                float4 current_vec = *(float4*)(&a[i]);
                min_vec = current_vec < min_vec ? current_vec : min_vec;
            }
            
            // Horizontal reduction of min_vec
            float temp_min = min_vec.x;
            if (min_vec.y < temp_min) temp_min = min_vec.y;
            if (min_vec.z < temp_min) temp_min = min_vec.z;
            if (min_vec.w < temp_min) temp_min = min_vec.w;
            
            if (temp_min < min_val) min_val = temp_min;
        }
        
        // Scalar tail
        for (; i < LEN_1D; i++) {
            float current = a[i];
            if (current < min_val) {
                min_val = current;
            }
        }
        
        x = min_val;
    }
    
    return x;
}
