#include <stddef.h>

void vectorized_s322(int iterations, int LEN_1D, float* a, float* b, float* c) {
    if (LEN_1D <= 2 || iterations <= 0) return;
    
    int outer_iters = iterations / 2;
    int inner_len = LEN_1D - 2;
    
    for (int nl = 0; nl < outer_iters; nl++) {
        // Store original values for the first two elements
        float a0 = a[0];
        float a1 = a[1];
        
        // Process elements with dependency chain
        for (int i = 0; i < inner_len; i++) {
            int idx = i + 2;
            // Compute new value using updated dependencies
            float new_val = a[idx] + a1 * b[idx] + a0 * c[idx];
            
            // Shift dependencies for next iteration
            a0 = a1;       // a[i-2] becomes old a[i-1]
            a1 = new_val;  // a[i-1] becomes updated a[i]
            
            // Update array with new value
            a[idx] = new_val;
        }
    }
}
