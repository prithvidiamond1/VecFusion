void vectorized_s258(int iterations, float* a, float* b, float* c, float* d, float* e, float aa[256][256])
{
    float s;
    for (int nl = 0; nl < iterations; nl++) {
        s = 0.f;
        for (int i = 0; i < 256; ++i) {
            if (a[i] > 0.f) {
                s = d[i] * d[i];
            }
            b[i] = s * c[i] + d[i];
            e[i] = (s + 1.f) * aa[0][i];
        }
    }
}
