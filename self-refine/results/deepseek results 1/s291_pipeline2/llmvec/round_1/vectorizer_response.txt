#include <stdint.h>

void vectorized_s291(int iterations, int LEN_1D, float* a, float* b)
{
    int im1;
    for (int nl = 0; nl < 2 * iterations; nl++) {
        im1 = LEN_1D - 1;
        // Handle first element separately
        a[0] = (b[0] + b[im1]) * 0.5f;

        // Vectorize the main loop: a[i] = (b[i] + b[i-1]) * 0.5f for i in [1, LEN_1D)
        int i = 1;
        int limit = LEN_1D - 4;

        for (; i <= limit; i += 4) {
            a[i]     = (b[i]     + b[i - 1]) * 0.5f;
            a[i + 1] = (b[i + 1] + b[i])     * 0.5f;
            a[i + 2] = (b[i + 2] + b[i + 1]) * 0.5f;
            a[i + 3] = (b[i + 3] + b[i + 2]) * 0.5f;
        }

        // Scalar cleanup tail
        for (; i < LEN_1D; i++) {
            a[i] = (b[i] + b[i - 1]) * 0.5f;
        }
    }
}
