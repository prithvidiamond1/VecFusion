void vectorized_s1213(int iterations, int LEN_1D, float* a, float* b, float* c, float* d)
{
    // This loop has a loop-carried dependency through a[] and b[],
    // so we cannot fully vectorize the inner loop without breaking semantics.
    // We preserve the scalar semantics exactly but unroll by 4 for minor benefit.
    for (int nl = 0; nl < iterations; nl++) {
        int i = 1;
        int limit = LEN_1D - 1;

        // Unrolled by 4, but must be sequential due to dependencies
        for (; i <= limit - 4; i += 4) {
            a[i]   = b[i-1]   + c[i];
            b[i]   = a[i+1]   * d[i];

            a[i+1] = b[i]     + c[i+1];
            b[i+1] = a[i+2]   * d[i+1];

            a[i+2] = b[i+1]   + c[i+2];
            b[i+2] = a[i+3]   * d[i+2];

            a[i+3] = b[i+2]   + c[i+3];
            b[i+3] = a[i+4]   * d[i+3];
        }

        // Scalar cleanup
        for (; i < limit; i++) {
            a[i] = b[i-1] + c[i];
            b[i] = a[i+1] * d[i];
        }
    }
}
