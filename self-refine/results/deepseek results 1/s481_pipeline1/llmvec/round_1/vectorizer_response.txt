#include <stdint.h>

void vectorized_s481(int iterations, int LEN_1D, float* a, float* b, float* c, float* d)
{
    for (int nl = 0; nl < iterations; nl++) {
        int i = 0;
        // Process 4 elements at a time
        for (; i <= LEN_1D - 4; i += 4) {
            // Check d[i..i+3] for negative values
            if (d[i] < 0.0f || d[i+1] < 0.0f || d[i+2] < 0.0f || d[i+3] < 0.0f) {
                // Find the exact first negative and return
                for (int j = i; j < i + 4; j++) {
                    if (d[j] < 0.0f) {
                        // But we need to apply updates for i..j-1 first
                        // Actually the scalar version returns immediately without updating a[j]
                        // We already updated a[i..i-1] in previous iterations, so just return
                        return;
                    }
                    a[j] += b[j] * c[j];
                }
                return;
            }
            a[i]   += b[i]   * c[i];
            a[i+1] += b[i+1] * c[i+1];
            a[i+2] += b[i+2] * c[i+2];
            a[i+3] += b[i+3] * c[i+3];
        }
        // Scalar cleanup tail
        for (; i < LEN_1D; i++) {
            if (d[i] < 0.0f) {
                return;
            }
            a[i] += b[i] * c[i];
        }
    }
}
