#include <stdlib.h>

void vectorized_s241(int iterations, int LEN_1D, float* a, float* b, float* c, float* d)
{
    float* a_old = (float*)malloc(LEN_1D * sizeof(float));

    for (int nl = 0; nl < 2*iterations; nl++) {
        // Store original a values before modification
        int i = 0;
        for (; i <= LEN_1D - 8; i += 8) {
            a_old[i+0] = a[i+0];
            a_old[i+1] = a[i+1];
            a_old[i+2] = a[i+2];
            a_old[i+3] = a[i+3];
            a_old[i+4] = a[i+4];
            a_old[i+5] = a[i+5];
            a_old[i+6] = a[i+6];
            a_old[i+7] = a[i+7];
        }
        for (; i < LEN_1D; i++) {
            a_old[i] = a[i];
        }

        // First compute all a[i] values (LEN_1D-1 elements)
        int n = LEN_1D - 1;
        i = 0;
        for (; i <= n - 8; i += 8) {
            a[i+0] = b[i+0] * c[i+0] * d[i+0];
            a[i+1] = b[i+1] * c[i+1] * d[i+1];
            a[i+2] = b[i+2] * c[i+2] * d[i+2];
            a[i+3] = b[i+3] * c[i+3] * d[i+3];
            a[i+4] = b[i+4] * c[i+4] * d[i+4];
            a[i+5] = b[i+5] * c[i+5] * d[i+5];
            a[i+6] = b[i+6] * c[i+6] * d[i+6];
            a[i+7] = b[i+7] * c[i+7] * d[i+7];
        }
        for (; i < n; i++) {
            a[i] = b[i] * c[i] * d[i];
        }

        // Then compute all b[i] values using stored a_old[i+1]
        i = 0;
        for (; i <= n - 8; i += 8) {
            b[i+0] = a[i+0] * a_old[i+1] * d[i+0];
            b[i+1] = a[i+1] * a_old[i+2] * d[i+1];
            b[i+2] = a[i+2] * a_old[i+3] * d[i+2];
            b[i+3] = a[i+3] * a_old[i+4] * d[i+3];
            b[i+4] = a[i+4] * a_old[i+5] * d[i+4];
            b[i+5] = a[i+5] * a_old[i+6] * d[i+5];
            b[i+6] = a[i+6] * a_old[i+7] * d[i+6];
            b[i+7] = a[i+7] * a_old[i+8] * d[i+7];
        }
        for (; i < n; i++) {
            b[i] = a[i] * a_old[i+1] * d[i];
        }
    }

    free(a_old);
}
