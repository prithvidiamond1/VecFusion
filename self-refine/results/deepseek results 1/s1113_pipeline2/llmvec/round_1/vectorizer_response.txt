#include <stdint.h>

#ifndef LEN_1D
#define LEN_1D 32000
#endif

void vectorized_s1113(float *a, float *b) {
    float scalar_val = a[LEN_1D / 2];

    typedef float float4 __attribute__((vector_size(16)));

    int i = 0;
    int limit = (LEN_1D / 4) * 4;

    float4 sv = {scalar_val, scalar_val, scalar_val, scalar_val};

    for (; i < limit; i += 4) {
        float4 bv;
        bv[0] = b[i + 0];
        bv[1] = b[i + 1];
        bv[2] = b[i + 2];
        bv[3] = b[i + 3];

        float4 rv = sv + bv;

        a[i + 0] = rv[0];
        a[i + 1] = rv[1];
        a[i + 2] = rv[2];
        a[i + 3] = rv[3];
    }

    for (; i < LEN_1D; i++) {
        a[i] = scalar_val + b[i];
    }
}
