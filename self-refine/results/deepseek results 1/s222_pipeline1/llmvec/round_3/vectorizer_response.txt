#include <stdint.h>

void vectorized_s222(int iterations, int LEN_1D, float*a, float*b, float*c, float* e)
{
    for (int nl = 0; nl < iterations; nl++) {
        // Vectorizable: a[i] += b[i] * c[i], starting at i=1
        int i = 1;
        for (; i <= LEN_1D - 8; i += 8) {
            a[i+0] += b[i+0] * c[i+0];
            a[i+1] += b[i+1] * c[i+1];
            a[i+2] += b[i+2] * c[i+2];
            a[i+3] += b[i+3] * c[i+3];
            a[i+4] += b[i+4] * c[i+4];
            a[i+5] += b[i+5] * c[i+5];
            a[i+6] += b[i+6] * c[i+6];
            a[i+7] += b[i+7] * c[i+7];
        }
        for (; i < LEN_1D; i++) {
            a[i] += b[i] * c[i];
        }

        // Scalar: loop-carried dependency on e[]
        for (i = 1; i < LEN_1D; i++) {
            e[i] = e[i - 1] * e[i - 1];
        }

        // Vectorizable: a[i] -= b[i] * c[i], starting at i=1
        i = 1;
        for (; i <= LEN_1D - 8; i += 8) {
            a[i+0] -= b[i+0] * c[i+0];
            a[i+1] -= b[i+1] * c[i+1];
            a[i+2] -= b[i+2] * c[i+2];
            a[i+3] -= b[i+3] * c[i+3];
            a[i+4] -= b[i+4] * c[i+4];
            a[i+5] -= b[i+5] * c[i+5];
            a[i+6] -= b[i+6] * c[i+6];
            a[i+7] -= b[i+7] * c[i+7];
        }
        for (; i < LEN_1D; i++) {
            a[i] -= b[i] * c[i];
        }
    }
}
