typedef float float4 __attribute__((ext_vector_type(4)));

float vectorized_s319(int iterations, int LEN_1D, float *a, float *b, float *c, float *d, float *e) {
    float sum = 0.0f;
    int outer_iterations = 2 * iterations;

    for (int nl = 0; nl < outer_iterations; nl++) {
        float sum_a = 0.0f;
        float sum_b = 0.0f;

        // First loop: a[i] = c[i] + d[i], sum_a += a[i]
        int i = 0;
        for (; i + 3 < LEN_1D; i += 4) {
            float4 c_vec = *(float4 *)&c[i];
            float4 d_vec = *(float4 *)&d[i];
            float4 a_vec = c_vec + d_vec;
            *(float4 *)&a[i] = a_vec;
            
            // Preserve scalar accumulation order for exact match
            sum_a += a_vec.x;
            sum_a += a_vec.y;
            sum_a += a_vec.z;
            sum_a += a_vec.w;
        }
        for (; i < LEN_1D; i++) {
            a[i] = c[i] + d[i];
            sum_a += a[i];
        }

        // Second loop: b[i] = c[i] + e[i], sum_b += b[i]
        i = 0;
        for (; i + 3 < LEN_1D; i += 4) {
            float4 c_vec = *(float4 *)&c[i];
            float4 e_vec = *(float4 *)&e[i];
            float4 b_vec = c_vec + e_vec;
            *(float4 *)&b[i] = b_vec;
            
            // Preserve scalar accumulation order for exact match
            sum_b += b_vec.x;
            sum_b += b_vec.y;
            sum_b += b_vec.z;
            sum_b += b_vec.w;
        }
        for (; i < LEN_1D; i++) {
            b[i] = c[i] + e[i];
            sum_b += b[i];
        }

        sum = sum_a + sum_b;
    }

    return sum;
}
