float vectorized_s352(int iterations, int LEN_1D, float* a, float *b) {
    float dot = 0.0f;

    for (int nl = 0; nl < 8 * iterations; nl++) {
        float partial_dot = 0.0f;
        
        // Process 8 elements at a time with immediate horizontal sum
        int i = 0;
        int main_len = LEN_1D - (LEN_1D % 8);
        
        for (; i < main_len; i += 8) {
            typedef float v8f __attribute__((vector_size(32), aligned(1)));
            v8f va = *(v8f*)&a[i];
            v8f vb = *(v8f*)&b[i];
            v8f vprod = va * vb;
            
            // Horizontal sum of this vector's products
            float* prod_ptr = (float*)&vprod;
            partial_dot += prod_ptr[0] + prod_ptr[1] + prod_ptr[2] + prod_ptr[3] +
                           prod_ptr[4] + prod_ptr[5] + prod_ptr[6] + prod_ptr[7];
        }
        
        // Scalar tail for remaining elements
        for (; i < LEN_1D; i++) {
            partial_dot += a[i] * b[i];
        }
        
        // Accumulate across outer iterations
        dot += partial_dot;
    }
    
    return dot;
}
