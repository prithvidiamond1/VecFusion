float vectorized_s315(int iterations, int LEN_1D, float* a)
{

        float max0 = a[0], max1 = -FLT_MAX, max2 = -FLT_MAX, max3 = -FLT_MAX;

            if (v3 > max3) max3 = v3;
        }
        for (; i < LEN_1D; i++) {
            float v = a[i];

        // Phase 2: Find index of max_val using 4-wide unroll
        int max_idx = 0;
        int idx0 = 0, idx1 = 0, idx2 = 0, idx3 = 0;

        if (idx3 > max_idx) max_idx = idx3;

        x = max_val;
        index = max_idx;
    }
