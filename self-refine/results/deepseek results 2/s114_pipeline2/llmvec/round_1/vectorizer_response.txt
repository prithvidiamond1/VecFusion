#include <string.h>

void vectorized_s114(int iterations, float aa[256][256], float bb[256][256])
{
    int loop_count = 200 * (iterations / 256);

    for (int nl = 0; nl < loop_count; nl++) {
        for (int i = 1; i < 256; i++) {
            float upper_values[256];

            // Vectorized gather of upper triangle values
            int j = 0;
            for (; j <= i - 4; j += 4) {
                upper_values[j]   = aa[j][i];
                upper_values[j+1] = aa[j+1][i];
                upper_values[j+2] = aa[j+2][i];
                upper_values[j+3] = aa[j+3][i];
            }
            for (; j < i; j++) {
                upper_values[j] = aa[j][i];
            }

            // Vectorized computation and write of lower triangle values
            j = 0;
            for (; j <= i - 8; j += 8) {
                aa[i][j]   = upper_values[j]   + bb[i][j];
                aa[i][j+1] = upper_values[j+1] + bb[i][j+1];
                aa[i][j+2] = upper_values[j+2] + bb[i][j+2];
                aa[i][j+3] = upper_values[j+3] + bb[i][j+3];
                aa[i][j+4] = upper_values[j+4] + bb[i][j+4];
                aa[i][j+5] = upper_values[j+5] + bb[i][j+5];
                aa[i][j+6] = upper_values[j+6] + bb[i][j+6];
                aa[i][j+7] = upper_values[j+7] + bb[i][j+7];
            }
            for (; j < i; j++) {
                aa[i][j] = upper_values[j] + bb[i][j];
            }
        }
    }
}
