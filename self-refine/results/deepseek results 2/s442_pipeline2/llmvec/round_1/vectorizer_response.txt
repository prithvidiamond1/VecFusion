#include <stdint.h>

void vectorized_s442(int iterations, int LEN_1D, float* a, float *b, float* c, float *d, float* e, int* indx)
{
    // Precompute indices for each case
    int idx1[LEN_1D], idx2[LEN_1D], idx3[LEN_1D], idx4[LEN_1D];
    int cnt1 = 0, cnt2 = 0, cnt3 = 0, cnt4 = 0;

    for (int i = 0; i < LEN_1D; i++) {
        switch (indx[i]) {
            case 1: idx1[cnt1++] = i; break;
            case 2: idx2[cnt2++] = i; break;
            case 3: idx3[cnt3++] = i; break;
            case 4: idx4[cnt4++] = i; break;
        }
    }

    for (int nl = 0; nl < iterations/2; nl++) {
        // Process case 1: a[i] += b[i] * b[i]
        {
            int j = 0;
            int limit = cnt1 - (cnt1 % 4);
            for (; j < limit; j += 4) {
                int i0 = idx1[j+0];
                int i1 = idx1[j+1];
                int i2 = idx1[j+2];
                int i3 = idx1[j+3];
                a[i0] += b[i0] * b[i0];
                a[i1] += b[i1] * b[i1];
                a[i2] += b[i2] * b[i2];
                a[i3] += b[i3] * b[i3];
            }
            for (; j < cnt1; j++) {
                int i = idx1[j];
                a[i] += b[i] * b[i];
            }
        }

        // Process case 2: a[i] += c[i] * c[i]
        {
            int j = 0;
            int limit = cnt2 - (cnt2 % 4);
            for (; j < limit; j += 4) {
                int i0 = idx2[j+0];
                int i1 = idx2[j+1];
                int i2 = idx2[j+2];
                int i3 = idx2[j+3];
                a[i0] += c[i0] * c[i0];
                a[i1] += c[i1] * c[i1];
                a[i2] += c[i2] * c[i2];
                a[i3] += c[i3] * c[i3];
            }
            for (; j < cnt2; j++) {
                int i = idx2[j];
                a[i] += c[i] * c[i];
            }
        }

        // Process case 3: a[i] += d[i] * d[i]
        {
            int j = 0;
            int limit = cnt3 - (cnt3 % 4);
            for (; j < limit; j += 4) {
                int i0 = idx3[j+0];
                int i1 = idx3[j+1];
                int i2 = idx3[j+2];
                int i3 = idx3[j+3];
                a[i0] += d[i0] * d[i0];
                a[i1] += d[i1] * d[i1];
                a[i2] += d[i2] * d[i2];
                a[i3] += d[i3] * d[i3];
            }
            for (; j < cnt3; j++) {
                int i = idx3[j];
                a[i] += d[i] * d[i];
            }
        }

        // Process case 4: a[i] += e[i] * e[i]
        {
            int j = 0;
            int limit = cnt4 - (cnt4 % 4);
            for (; j < limit; j += 4) {
                int i0 = idx4[j+0];
                int i1 = idx4[j+1];
                int i2 = idx4[j+2];
                int i3 = idx4[j+3];
                a[i0] += e[i0] * e[i0];
                a[i1] += e[i1] * e[i1];
                a[i2] += e[i2] * e[i2];
                a[i3] += e[i3] * e[i3];
            }
            for (; j < cnt4; j++) {
                int i = idx4[j];
                a[i] += e[i] * e[i];
            }
        }
    }
}
