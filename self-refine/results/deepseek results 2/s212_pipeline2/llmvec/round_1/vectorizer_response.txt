#include <stddef.h>

void vectorized_s212(int n, float *a, float *b, float *c, float *d) {
    int limit = n - 1;
    if (limit <= 0) return;

    // First loop: b[i] += a[i+1] * d[i]
    // Vectorize with width 8
    int i = 0;
    int vec_limit = limit - (limit % 8);
    for (; i < vec_limit; i += 8) {
        b[i+0] += a[i+1] * d[i+0];
        b[i+1] += a[i+2] * d[i+1];
        b[i+2] += a[i+3] * d[i+2];
        b[i+3] += a[i+4] * d[i+3];
        b[i+4] += a[i+5] * d[i+4];
        b[i+5] += a[i+6] * d[i+5];
        b[i+6] += a[i+7] * d[i+6];
        b[i+7] += a[i+8] * d[i+7];
    }
    for (; i < limit; i++) {
        b[i] += a[i+1] * d[i];
    }

    // Second loop: a[i] *= c[i]
    i = 0;
    for (; i < vec_limit; i += 8) {
        a[i+0] *= c[i+0];
        a[i+1] *= c[i+1];
        a[i+2] *= c[i+2];
        a[i+3] *= c[i+3];
        a[i+4] *= c[i+4];
        a[i+5] *= c[i+5];
        a[i+6] *= c[i+6];
        a[i+7] *= c[i+7];
    }
    for (; i < limit; i++) {
        a[i] *= c[i];
    }
}
