void vectorized_s322(int iterations, int LEN_1D, float* a, float *b, float* c)
{
    // This loop has a loop-carried dependency through a[i-1] and a[i-2],
    // so true SIMD vectorization across i is not safe.
    // We can only vectorize the outer nl loop if arrays are independent across iterations,
    // but they are not (a is modified each iteration).
    // Therefore, we unroll the inner loop manually with scalar operations,
    // keeping the same semantics but with some loop unrolling for ILP.
    for (int nl = 0; nl < iterations/2; nl++) {
        int i = 2;
        // Unroll by 4, but must be careful: each iteration depends on previous two
        // So we cannot truly parallelize, but we can unroll to help the compiler pipeline
        for (; i < LEN_1D - 3; i += 4) {
            // These must be computed sequentially due to dependency
            float a0 = a[i]   + a[i-1]   * b[i]   + a[i-2]   * c[i];
            a[i] = a0;
            float a1 = a[i+1] + a[i]     * b[i+1] + a[i-1]   * c[i+1];
            a[i+1] = a1;
            float a2 = a[i+2] + a[i+1]   * b[i+2] + a[i]     * c[i+2];
            a[i+2] = a2;
            float a3 = a[i+3] + a[i+2]   * b[i+3] + a[i+1]   * c[i+3];
            a[i+3] = a3;
        }
        // Scalar cleanup
        for (; i < LEN_1D; i++) {
            a[i] = a[i] + a[i-1] * b[i] + a[i-2] * c[i];
        }
    }
}
