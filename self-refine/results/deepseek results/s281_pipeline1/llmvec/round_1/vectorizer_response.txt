void vectorized_s281(int iterations, int LEN_1D, float* a, float* b, float* c)
{
    // Note: this loop has a loop-carried dependency through a[] (a[LEN_1D-i-1] reads
    // from positions that overlap with a[i] writes), so we cannot vectorize the inner
    // loop in general. We preserve scalar semantics exactly.
    float x;
    for (int nl = 0; nl < iterations; nl++) {
        for (int i = 0; i < LEN_1D; i++) {
            x = a[LEN_1D - i - 1] + b[i] * c[i];
            a[i] = x - (float)1.0;
            b[i] = x;
        }
    }
}
